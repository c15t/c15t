<import src="../../../shared/concepts/client-modes.mdx#hosted-mode" />

```ts
import { createConsentKernel, createHostedTransport } from 'c15t';

const kernel = createConsentKernel({
  transport: createHostedTransport({
    backendURL: 'https://your-instance.c15t.dev',
  }),
});

await kernel.commands.init();
```

<import src="../../../shared/concepts/client-modes.mdx#offline-mode" />

```ts
import { createConsentKernel, createOfflineTransport } from 'c15t';

const kernel = createConsentKernel({
  initialOverrides: {
    country: 'DE', // Override country to trigger GDPR jurisdiction
  },
  transport: createOfflineTransport(),
});

await kernel.commands.init();
```

<import src="../../../shared/concepts/client-modes.mdx#custom-mode" />

```ts
import { createConsentKernel, type KernelTransport } from 'c15t';

const transport: KernelTransport = {
  async init() {
    const response = await fetch('/my-api/consent/init');
    if (!response.ok) throw new Error('Consent initialization failed');
    return response.json();
  },
  async save(payload) {
    const response = await fetch('/my-api/consent', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload),
    });
    return { ok: response.ok };
  },
};

const kernel = createConsentKernel({
  transport,
});

await kernel.commands.init();
```
