Subscribe to kernel events before initializing to observe the full lifecycle:

```ts
import { createConsentKernel, createHostedTransport } from 'c15t';

const kernel = createConsentKernel({
  transport: createHostedTransport({
    backendURL: 'https://your-instance.c15t.dev',
  }),
});

kernel.events.on('init:applied', ({ snapshot }) => {
  console.log('Init complete:', snapshot.location);
});

kernel.events.on('command:save:completed', ({ result }) => {
  console.log('Consent save complete:', result);
});

kernel.events.on('command:error', ({ command, error }) => {
  console.error(`${command} failed`, error);
});

await kernel.commands.init();
```
