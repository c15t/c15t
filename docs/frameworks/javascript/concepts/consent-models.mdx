Read the current consent model from the kernel snapshot:

```ts
import { createConsentKernel, createHostedTransport } from 'c15t';

const kernel = createConsentKernel({
  transport: createHostedTransport({
    backendURL: 'https://your-instance.c15t.dev',
  }),
});

await kernel.commands.init();

const { model, policy } = kernel.getSnapshot();
console.log('Current consent model:', model ?? 'detecting...');
if (policy) {
  console.log('From policy:', policy.id);
}

// React to model changes
let previousModel = model;
kernel.subscribe((snapshot) => {
  if (snapshot.model !== previousModel) {
    console.log('Model resolved:', snapshot.model);
    previousModel = snapshot.model;
  }
});
```

Override the detected jurisdiction for testing:

```ts
// Test as EU visitor
kernel.set.overrides({ country: 'DE' });
await kernel.commands.init();

// Test as California visitor
kernel.set.overrides({ country: 'US', region: 'CA' });
await kernel.commands.init();
```

<Callout type="info">
  For full control over which model applies where — plus UI customization, category scoping, and re-prompting — see [Policy Packs](/docs/frameworks/javascript/concepts/policy-packs).
</Callout>
