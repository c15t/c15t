## Quick Setup

Bind the IAB module to an initialized kernel:

```ts
import { createConsentKernel, createHostedTransport } from 'c15t';
import { createIAB } from '@c15t/iab';

const kernel = createConsentKernel({
  transport: createHostedTransport({
    backendURL: 'https://your-instance.c15t.dev',
  }),
});

await kernel.commands.init();

const iab = createIAB({
  kernel,
  cmpId: 123,
  vendors: [1, 2, 10, 25],
});
```

## Accessing IAB State

IAB state is available on the kernel snapshot when IAB is enabled:

```ts
const { iab: iabState } = kernel.getSnapshot();

if (iabState) {
  console.log('GVL:', iabState.gvl);
  console.log('Vendor consents:', iabState.vendorConsents);
  console.log('Purpose consents:', iabState.purposeConsents);
}

// Subscribe to IAB state changes
let previousIab = iabState;
kernel.subscribe((snapshot) => {
  if (snapshot.iab !== previousIab && snapshot.iab) {
    console.log('IAB state updated:', snapshot.iab);
    previousIab = snapshot.iab;
  }
});
```

## `__tcfapi` Stub

When IAB is enabled, c15t automatically registers the `__tcfapi` function on the window object. This is the standard CMP API that ad tech vendors use to check consent:

```ts
// Third-party scripts can use the standard TCF API
window.__tcfapi('getTCData', 2, (tcData, success) => {
  if (success) {
    console.log('TC String:', tcData.tcString);
    console.log('Purpose consents:', tcData.purpose.consents);
  }
});
```
