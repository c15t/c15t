## Configuration

Create the blocker after initializing the kernel:

```ts
import { createConsentKernel, createHostedTransport } from 'c15t';
import { createNetworkBlocker } from 'c15t/modules/network-blocker';

const kernel = createConsentKernel({
  transport: createHostedTransport({
    backendURL: 'https://your-instance.c15t.dev',
  }),
});

await kernel.commands.init();

const networkBlocker = createNetworkBlocker({
  kernel,
  rules: [
      {
        id: 'google-analytics',
        domain: 'google-analytics.com',
        category: 'measurement',
      },
      {
        id: 'facebook-pixel',
        domain: 'facebook.com',
        pathIncludes: '/tr',
        category: 'marketing',
      },
      {
        id: 'analytics-post',
        domain: 'analytics.example.com',
        methods: ['POST'],
        category: 'measurement',
      },
  ],
});
```

## Runtime Updates

Update the network blocker configuration at runtime:

```ts
networkBlocker.updateRules([
    {
      id: 'new-tracker',
      domain: 'tracker.example.com',
      category: 'marketing',
    },
]);

// Restore fetch and XMLHttpRequest during teardown
networkBlocker.dispose();
```
