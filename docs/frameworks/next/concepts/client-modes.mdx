```tsx
import { type ReactNode } from 'react';
import { hosted, ConsentProvider, ConsentBanner } from 'c15t/next';

export function ConsentManager({ children }: { children: ReactNode }) {
  return (
    <ConsentProvider
      options={{
        mode: hosted({ url: '/api/c15t' }),
      }}
    >
      <ConsentBanner />
      {children}
    </ConsentProvider>
  );
}
```

<import src="../../../shared/concepts/client-modes.mdx#offline-mode" />

```tsx
import { type ReactNode } from 'react';
import { offline, ConsentProvider, ConsentBanner } from 'c15t/next';

export function ConsentManager({ children }: { children: ReactNode }) {
  return (
    <ConsentProvider
      options={{
        mode: offline(),
        overrides: {
          country: 'DE', // Override country to trigger GDPR jurisdiction
        },
      }}
    >
      <ConsentBanner />
      {children}
    </ConsentProvider>
  );
}
```

<import src="../../../shared/concepts/client-modes.mdx#custom-mode" />

```tsx
import { type ReactNode } from 'react';
import { custom, ConsentProvider, ConsentBanner } from 'c15t/next';

export function ConsentManager({ children }: { children: ReactNode }) {
  return (
    <ConsentProvider
      options={{
        mode: custom({
          async init() {
            const res = await fetch('/my-api/consent/init');
            const data = await res.json();
            return { data, ok: res.ok };
          },
          async setConsent(options) {
            const res = await fetch('/my-api/consent', {
              method: 'POST',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify(options?.body),
            });
            return { data: await res.json(), ok: res.ok };
          },
        }),
      }}
    >
      <ConsentBanner />
      {children}
    </ConsentProvider>
  );
}
```
