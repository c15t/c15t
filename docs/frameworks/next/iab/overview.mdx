## Prepare the IAB configuration

<import src="../../../shared/react/iab/overview.mdx#prepare" />

Next.js uses the React IAB components. There is no `c15t/next/iab` JavaScript
entry point, so import the components from `c15t/react/iab` and render this
Client Component inside the `ConsentRoot` from your
[App Router](/docs/frameworks/next/app-router) or
[Pages Router](/docs/frameworks/next/pages-router) setup. Import
`c15t/next/iab/styles.css` at your global stylesheet entry, after
`c15t/next/styles.css`: the IAB stylesheet holds only the IAB rules and uses
the tokens from the base stylesheet.

```tsx title="components/iab-consent.tsx"
'use client';

import { IABProvider, IABConsentBanner, IABConsentDialog } from 'c15t/react/iab';

export function IABConsent({ cmpId }: { cmpId: number }) {
  return (
    <IABProvider cmpId={cmpId}>
      <IABConsentBanner />
      <IABConsentDialog />
    </IABProvider>
  );
}
```

## Restrict how vendors use a purpose

Publisher restrictions override a vendor's declared use of one purpose. Pass
them to `IABProvider` as `publisherRestrictions`. The vendor IDs below are
placeholders; use vendors from your list whose declarations allow each
restriction.

```tsx title="components/iab-consent.tsx"
'use client';

import { IABProvider, IABConsentBanner, IABConsentDialog } from 'c15t/react/iab';

const publisherRestrictions = [
  // Vendors 10 and 755 may not use purpose 4.
  { purposeId: 4, restrictionType: 0 as const, vendorIds: [10, 755] },
  // Vendor 755 must use legitimate interest for purpose 7.
  { purposeId: 7, restrictionType: 2 as const, vendorIds: [755] },
];

export function IABConsent({ cmpId }: { cmpId: number }) {
  return (
    <IABProvider cmpId={cmpId} publisherRestrictions={publisherRestrictions}>
      <IABConsentBanner />
      <IABConsentDialog />
    </IABProvider>
  );
}
```
