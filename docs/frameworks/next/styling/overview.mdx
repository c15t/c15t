## Quick Start

```tsx
import { hosted } from '@c15t/nextjs';
import {
  type ReactUIOptions,
  ConsentProvider,
  ConsentBanner,
  ConsentDialog,
} from 'c15t/next';
type Theme = NonNullable<ReactUIOptions['theme']>;

const theme = {
  colors: {
    primary: '#6366f1',
    primaryHover: '#4f46e5',
  },
  radius: {
    md: '0.75rem',
    lg: '1rem',
  },
  slots: {
    consentBannerTitle: 'text-xl font-semibold',
    buttonPrimary: 'rounded-full',
  },
} satisfies Theme;

export function ConsentManager({ children }) {
  return (
    <ConsentProvider
      options={{
        mode: hosted({ url: '/api/c15t' }),
        theme,
      }}
    >
      <ConsentBanner />
      <ConsentDialog />
      {children}
    </ConsentProvider>
  );
}
```
