## Basic Usage

```tsx
import { hosted } from '@c15t/nextjs';
import { ConsentProvider, ConsentBanner } from 'c15t/next';

export function ConsentManager() {
  return (
    <ConsentProvider options={{ mode: hosted({ url: '/api/c15t' }) }}>
      <ConsentBanner />
    </ConsentProvider>
  );
}
```
