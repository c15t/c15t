## Usage

```tsx
'use client';

import dynamic from 'next/dynamic';
import { type ReactNode } from 'react';
import { ConsentProvider } from 'c15t/next/v3';

const DevTools =
  process.env.NODE_ENV === 'development'
    ? dynamic(() => import('c15t/next/v3/devtools').then(({ DevTools }) => DevTools), {
        ssr: false,
      })
    : () => null;

export function ConsentManager({ children }: { children: ReactNode }) {
  return (
    <ConsentProvider options={{ mode: 'hosted', backendURL: '/api/c15t' }}>
      {children}
      <DevTools />
    </ConsentProvider>
  );
}
```

For an App Router setup with server prefetching, put `DevTools` inside `ConsentBoundary` so it receives the boundary's kernel. Import the adapter from `c15t/next/v3/devtools`, or `@c15t/nextjs/v3/devtools` when you install the Next.js package directly.
