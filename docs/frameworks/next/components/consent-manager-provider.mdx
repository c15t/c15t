---
title: ConsentBoundary
description: The Next.js boundary that hydrates a request-scoped consent kernel.
group: frameworks
---

Use `ConsentBoundary` when a Server Component supplies request-derived kernel
configuration. Use `ConsentProvider` directly in client-only routes.

```tsx title="app/layout.tsx"
import { ConsentBoundary } from 'c15t/next';
import { readInitialConsentConfig } from 'c15t/next/server';

export default async function RootLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  const config = await readInitialConsentConfig();

  return (
    <html lang="en">
      <body>
        <ConsentBoundary backendURL="/api/c15t" config={config}>
          {children}
        </ConsentBoundary>
      </body>
    </html>
  );
}
```
