## Using Tailwind with Component Slots

Apply Tailwind classes via `options.components`:

```tsx
import { hosted } from '@c15t/react';
import { type ReactNode } from 'react';
import { ConsentProvider } from 'c15t/react';

export function ConsentManager({ children }: { children: ReactNode }) {
  return (
    <ConsentProvider
      options={{
        mode: hosted({ url: 'https://your-instance.c15t.dev' }),
        components: {
          banner: {
            root: { className: 'fixed bottom-0 inset-x-0 z-50' },
            card: {
              className:
                'mx-auto max-w-2xl rounded-t-2xl bg-white p-6 shadow-2xl',
            },
            title: { className: 'text-lg font-semibold text-gray-900' },
            footer: { className: 'mt-4 flex flex-wrap gap-3' },
          },
          description: {
            banner: { className: 'mt-2 text-sm text-gray-600' },
          },
          button: {
            primary: {
              className:
                'rounded-full bg-indigo-600 px-6 py-2.5 text-sm font-medium text-white hover:bg-indigo-700',
            },
            secondary: {
              className:
                'rounded-full border border-gray-300 px-6 py-2.5 text-sm font-medium text-gray-700 hover:bg-gray-50',
            },
          },
        },
      }}
    >
      {children}
    </ConsentProvider>
  );
}
```
