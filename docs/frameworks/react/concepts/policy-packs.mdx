Inspect the resolved policy at runtime with `useConsentManager()`:

```tsx
import { useConsentManager } from '@c15t/react';

export function PolicyStatus() {
  const { policyRule, resolution, promptRequirement } = useConsentManager();
  return (
    <pre>
      {JSON.stringify(
        { policyRule, resolution: resolution.status, promptRequirement },
        null,
        2
      )}
    </pre>
  );
}
```
