Configure callbacks in the provider to log lifecycle events, and add DevTools for a visual inspector:

```tsx
import { hosted } from '@c15t/react';
import { type ReactNode } from 'react';
import { ConsentProvider, ConsentBanner, ConsentDialog } from 'c15t/react';
import { DevTools } from '@c15t/dev-tools/react';

export default function ConsentManager({ children }: { children: ReactNode }) {
  return (
    <ConsentProvider
      options={{
        mode: hosted({ url: 'https://your-instance.c15t.dev' }),
        callbacks: {
          onChoiceRecorded: ({ snapshot }) =>
            console.log(snapshot.explicitChoice),
          onPermissionsChanged: ({ snapshot }) =>
            console.log(snapshot.effectivePermissions),
        },
      }}
    >
      <ConsentBanner />
      <ConsentDialog />
      {process.env.NODE_ENV !== 'production' && <DevTools />}
      {children}
    </ConsentProvider>
  );
}
```
