## React Example (Hosted)

Point the provider at your backend — policy resolution happens server-side:

```tsx
import type { ReactNode } from 'react';
import {
  ConsentBanner,
  ConsentDialog,
  ConsentProvider,
  hosted,
} from 'c15t/react';

export function ConsentManager({ children }: { children: ReactNode }) {
  return (
    <ConsentProvider
      options={{
        mode: hosted({ url: 'https://your-instance.c15t.dev' }),
      }}
    >
      <ConsentBanner />
      <ConsentDialog />
      {children}
    </ConsentProvider>
  );
}
```

## Offline mode

For local development, previews, or automated tests, select offline mode and pass policies directly:

```tsx
import type { ReactNode } from 'react';
import {
  offline,
  ConsentBanner,
  ConsentDialog,
  ConsentProvider,
  policyPackPresets,
} from 'c15t/react';

export function ConsentManager({ children }: { children: ReactNode }) {
  return (
    <ConsentProvider
      options={{
        mode: offline(),
        offlinePolicy: {
          policyPacks: [
            policyPackPresets.europeOptIn(),
            policyPackPresets.californiaOptOut(),
            policyPackPresets.worldNoBanner(),
          ],
        },
        overrides: {
          country: 'DE', // Preview as a German visitor
        },
      }}
    >
      <ConsentBanner />
      <ConsentDialog />
      {children}
    </ConsentProvider>
  );
}
```
