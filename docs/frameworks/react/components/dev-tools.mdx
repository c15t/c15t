## Usage

```tsx
import { lazy, Suspense, type ReactNode } from 'react';
import { ConsentProvider } from 'c15t/react/v3';

const DevTools =
  process.env.NODE_ENV === 'development'
    ? lazy(() =>
        import('c15t/react/v3/devtools').then(({ DevTools }) => ({ default: DevTools })),
      )
    : () => null;

export function ConsentManager({ children }: { children: ReactNode }) {
  return (
    <ConsentProvider options={{ mode: 'hosted', backendURL: 'https://your-instance.c15t.dev' }}>
      {children}
      <Suspense fallback={null}>
        <DevTools />
      </Suspense>
    </ConsentProvider>
  );
}
```

Import the adapter from `c15t/react/v3/devtools` when you use the umbrella package, or `@c15t/react/v3/devtools` when you install the React package directly. `@c15t/dev-tools` exports the imperative engine, not the React component.
