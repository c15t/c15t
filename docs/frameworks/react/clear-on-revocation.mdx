## Configuration

Add `clearOnRevocation` to your provider options:

```tsx
import { type ReactNode } from 'react';
import { ConsentManagerProvider } from '@c15t/react';

export function ConsentManager({ children }: { children: ReactNode }) {
  return (
    <ConsentManagerProvider
      options={{
        mode: 'hosted',
        backendURL: 'https://your-instance.c15t.dev',
        clearOnRevocation: {
          marketing: {
            cookies: ['_fbq', '_fbp'],
          },
          measurement: {
            cookies: ['_ga', '_ga_*'],
            localStorage: ['ga_disable_*'],
          },
        },
      }}
    >
      {children}
    </ConsentManagerProvider>
  );
}
```

## Runtime Updates

Update the configuration at runtime - this immediately sweeps any category that is already denied:

```tsx
import { useConsentManager } from '@c15t/react';

function ClearOnRevocationManager() {
  const { setClearOnRevocation } = useConsentManager();

  const addRule = () => {
    setClearOnRevocation({
      marketing: {
        cookies: ['_fbq', '_fbp', 'new-tracker-cookie'],
      },
    });
  };

  return null;
}
```
